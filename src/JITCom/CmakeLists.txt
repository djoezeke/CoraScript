set(CORA_JITCOM_SOURCES 

    "${CORA_JITCOM_DIR}/JITEngine.cpp"
    "${CORA_JITCOM_DIR}/JITPipeline.cpp"
)

#--------------------------------------------------------------------
# Build JIT Compilation Library.
#--------------------------------------------------------------------

add_library(JITCom STATIC ${CORA_JITCOM_SOURCES})

add_library(Cora::JITCom ALIAS JITCom)

target_include_directories(JITCom
	PUBLIC
        ${PROJECT_SOURCE_DIR}/include
		${CORA_JITCOM_DIR}
)

target_link_libraries(JITCom
	PRIVATE
		Cora::AST
		Cora::Basic
		Cora::IRGen
		Cora::JITCom
		Cora::Parser
		Cora::Runtime
		Cora::Builtin
		Cora::Semantic
		Cora::VMachine
)
