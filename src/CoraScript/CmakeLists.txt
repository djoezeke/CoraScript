set(CORASCRIPT_SOURCES 
	"${CORASCRIPT_DIR}/main.cpp"
)

	PRIVATE

        ${PROJECT_SOURCE_DIR}/include
)

	PRIVATE
		Cora::Compiler
)

	target_link_libraries(Corascript
		PRIVATE
			Cora::Modules
	)
endif()
