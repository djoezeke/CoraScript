cmake_minimum_required(VERSION 3.16)

#--------------------------------------------------------------------
# Project.
#--------------------------------------------------------------------

project(CoraScript
        VERSION "0.1.0" 
        DESCRIPTION "Cora Scripting Language "
        HOMEPAGE_URL "https://github.com/djoezeke/CoraScript"
        LANGUAGES  C CXX
)

#--------------------------------------------------------------------
# Configure variables for the project.   
#--------------------------------------------------------------------

option(CORA_BUILD_COMPILER "Build Compiler library" ON)
option(CORA_BUILD_MODULES "Build external modules library" ON)
option(CORA_BUILD_CORASCRIPT "Build Corascript executable" ON)
option(CORA_BUILD_CORASCRIPTEN "Build CoraScripten executable" OFF)
option(CORA_CORASCRIPT_WITH_MODULES "Link Corascript with external modules library when available" ON)
option(CORA_BUILD_TESTS "Enable CTest-based project tests" ON)

set(MODULES_DIR "${CMAKE_CURRENT_SOURCE_DIR}/modules")
set(COMPILER_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/Compiler")
set(CORASCRIPT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/CoraScript")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

#--------------------------------------------------------------------
# Add project source directories.
#--------------------------------------------------------------------

add_subdirectory(src)

if(CORA_BUILD_TESTS)
        include(CTest)
        add_subdirectory(tests)
endif()
