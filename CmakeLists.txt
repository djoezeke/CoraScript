cmake_minimum_required(VERSION 3.16)

#--------------------------------------------------------------------
# Project.
#--------------------------------------------------------------------

project(CoraScript
        VERSION "0.1.0" 
        DESCRIPTION "Cora Scripting Language "
        HOMEPAGE_URL "https://github.com/djoezeke/CoraScript"
        LANGUAGES  C CXX
)

#--------------------------------------------------------------------
# Configure variables for the project.   
#--------------------------------------------------------------------

option(CORA_BUILD_CORA "Build cora executable" ON)
option(CORA_BUILD_CORAC "Build corac executable" OFF)
option(CORA_BUILD_CORAVM "Build coravm executable" ON)
option(CORA_CORASCRIPT_WITH_MODULES "Link Corascript with external modules library when available" ON)
option(CORA_BUILD_TESTS "Enable CTest-based project tests" ON)

set(CORA_AST_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/AST")
set(CORA_BASIC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/Basic")
set(CORA_BUILTIN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/Builtin")
set(CORA_CORA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/Cora")
set(CORA_IR_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/IR")
set(CORA_PARSER_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/Parser")
set(CORA_RUNTIME_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/Runtime")
set(CORA_VMACHINE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/VMachine")
set(CORA_MODULES_DIR "${CMAKE_CURRENT_SOURCE_DIR}/modules")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

#--------------------------------------------------------------------
# Add project source directories.
#--------------------------------------------------------------------

add_subdirectory(src)
add_subdirectory(modules)

if(CORA_BUILD_TESTS)
        include(CTest)
        add_subdirectory(tests)
endif()
